def check_password(password):
    Uppercase = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    Lowercase = "abcdefghijklmnopqrstuvwxyz"
    Numbers = "0123456789"
    Special = "!@#$%^&*"

    if len(password) < 4:
        print("Password is not Strong")
    else:
        upper = False
        lower = False
        number = False
        special = False

        for ch in password:
            if ch in Uppercase:
                upper = True
            elif ch in Lowercase:
                lower = True
            elif ch in Numbers:
                number = True
            elif ch in Special:
                special = True

        if upper and lower and number and special:
            print("Password is Strong")
        else:
            print("Password is not Strong")


password = input()
check_password(password)
